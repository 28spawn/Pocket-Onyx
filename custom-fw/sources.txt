# The source manifest - the ONLY place that says which modules each target
# compiles.
#
# The DEVICE compiles every .c under custom-fw/src: it is the whole firmware.
# The other targets compile the same tree MINUS the board files they replace
# themselves.
#
# This file exists because each build script used to carry its own hand-written
# list, and a new module was silently missing from one of them - which is how
# the emulator's build broke three times (pa_audio_core.c, pa_input.c, and the
# modules the Makefile and build.ps1 had but the two emu/*.ps1 scripts did not).
#
# tools/check_sources.py reads this file and fails the build if:
#   - an excluded path does not exist (a stale entry),
#   - a module is in no target at all (nobody builds it),
#   - a target excludes something that is not part of the board layer.
#
# Syntax
#   [name]                    a target, or a group if the name starts with "g_"
#   include = a, b, @grp      compile exactly these (globs and @groups allowed)
#   exclude = a, b            compile everything under src/ EXCEPT these
#
# Why each target excludes what it excludes:
#   src/board/main.c        the device's main(); each host has its own entry
#   src/board/audio_alnk.c  the ALNK0 driver (the host has its own I/O)
#   src/board/st7735s.c    the panel driver (the host draws into a buffer)
#   src/board/spi_nor.c     the NOR flash driver (the browser has a RAM image)
#   src/board/gpio_scan.c   the GPIO scanner (the browser has DOM buttons)
#   src/board/usb_ble.c    the USB/BLE driver (the browser has no USB audio)
#   src/sys/smp_cpu1.c      the dual-core offload, compiled but dead
#
# paths are relative to custom-fw/

# ---------------------------------------------------------------- the groups --
[g_dsp]

[g_nam]
include = src/nam/*.c

[g_fx]
include = src/fx/*.c

[g_sys]
include = src/sys/*.c

[g_board]
include = src/board/*.c

# ---------------------------------------------------------------- the targets --
